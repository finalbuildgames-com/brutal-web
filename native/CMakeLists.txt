cmake_minimum_required(VERSION 3.25)
project(BrutalRaylib LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_BUILD_RPATH "$ORIGIN")
if(EMSCRIPTEN)
 set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
else()
 set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
endif()
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GLFW_BUILD_WAYLAND OFF CACHE BOOL "" FORCE)
set(RAYLIB_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../build/deps/raylib" CACHE PATH "raylib 6.0 source")
set(BRUTAL_ADAPTER_ROOT "" CACHE PATH "Optional external adapters; never fetched by the public kit")
if(EMSCRIPTEN)
  set(PLATFORM Web CACHE STRING "" FORCE)
  set(OPENGL_VERSION "ES 3.0" CACHE STRING "" FORCE)
  add_compile_options(-fwasm-exceptions)
  if(BRUTAL_WEB_THREADS)
   add_compile_options(-pthread)
   add_link_options(-pthread)
  endif()
else()
  set(PLATFORM Desktop CACHE STRING "" FORCE)
  set(USE_EXTERNAL_GLFW OFF CACHE STRING "" FORCE)
endif()
add_subdirectory("${RAYLIB_SOURCE}" raylib EXCLUDE_FROM_ALL)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_BUILD_RPATH_USE_ORIGIN ON)
if(EMSCRIPTEN)
  add_library(brutal_raylib STATIC raylib_bridge.c loop.c wasm_sjlj.c sjlj_probe.c)
  set_target_properties(brutal_raylib PROPERTIES PREFIX "")
else()
  add_library(brutal_raylib SHARED raylib_bridge.c)
endif()
target_link_libraries(brutal_raylib PRIVATE raylib)
target_compile_options(brutal_raylib PRIVATE -Wall -Wextra -Werror)
target_include_directories(brutal_raylib PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(brutal_raylib PROPERTIES C_STANDARD 11)
include(physics.cmake)
if(BRUTAL_ADAPTER_ROOT)
  add_subdirectory("${BRUTAL_ADAPTER_ROOT}/native" adapters)
endif()

add_custom_target(brutal_all DEPENDS brutal_raylib brutal_physics)
if(BRUTAL_ADAPTER_ROOT)
 if(NOT TARGET "${BRUTAL_ADAPTER_TARGET}")
  message(FATAL_ERROR "Adapter must provide its declared library target")
 endif()
 add_dependencies(brutal_all "${BRUTAL_ADAPTER_TARGET}")
endif()
